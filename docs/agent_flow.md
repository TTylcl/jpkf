 状态枚举定义
 from enum import Enum

# 用户绑定状态
class BindingStatus(str, Enum):
    PENDING = "pending"      # 待验证
    ACTIVE = "active"        # 已绑定
    INACTIVE = "inactive"    # 已解绑

# 预排课状态
class PreScheduleStatus(str, Enum):
    PENDING = "pending"      # 待审核
    APPROVED = "approved"    # 已通过
    REJECTED = "rejected"    # 已拒绝
    CANCELLED = "cancelled"  # 已取消
    CONFIRMED = "confirmed"  # 已生成正式排课

# 请假状态
class LeaveStatus(str, Enum):
    PENDING = "pending"      # 待审核
    APPROVED = "approved"    # 已通过
    REJECTED = "rejected"    # 已拒绝
    CANCELLED = "cancelled"  # 已撤销
    EFFECTIVE = "effective"  # 已生效

# 会话状态
class SessionStatus(str, Enum):
    IDLE = "idle"                    # 空闲
    RECOGNIZING = "recognizing"      # 意图识别中
    VERIFYING = "verifying"          # 权限验证中
    EXECUTING = "executing"          # 执行业务中
    NEED_INPUT = "need_input"        # 等待用户输入
    REPLYING = "replying"            # 生成回复中
